from math import sqrt
input("Dis is the Right Angled triangle calculator (❁´◡`❁) enter your numbers")
#introductions
a = float(input("Enter your A value"))
#enter your A value (1)
b = float(input("Enter your B value "))
#enter your B value (2)
final = sqrt(pow(a, 2) + pow(b, 2))
print(f"The hypotenuse is {final:.2f}")
